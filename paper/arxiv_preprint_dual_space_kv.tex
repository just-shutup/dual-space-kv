\begin{abstract}
The linear memory footprint of the Key-Value (KV) cache is a primary bottleneck for deploying Large Language Models (LLMs) on memory-constrained devices. Existing eviction heuristics (e.g., H2O, StreamingLLM) prune tokens with low historical attention scores, but suffer from irreversible factual forgetting in long-context tasks (``Needle-In-A-Haystack''). Conversely, naive centroid clustering in physical Key-space fails ($\cos \le 0.53$) due to destructive phase dispersion induced by Rotary Position Embeddings (RoPE). 

In this paper, we analyze the geometric incompatibility between RoPE and key aggregation, proving that naive averaging leads to asymptotic norm collapse ($\|\bar{k}\| \to 0$). We propose \textbf{Windowed Dual-Space Centroid KV}, a position-constrained compression framework operating in a canonical coordinate space ($U$-space) via the inverse rotary mapping $u_t = R(-t) k_t$. By constraining clustering to local temporal windows ($W \le 32$) and re-rotating canonical centroids to the mean window position $\bar{k}_c = R(\bar{p}_c) \bar{u}_c$, our approach preserves vector magnitude without phase cancellation. Crucially, we conduct an exact byte-for-byte evaluation: each centroid requires only 260 bytes ($1.01\times$ of a vanilla token in FP16), eliminating hidden quadratic matrix overheads. On Qwen2.5-0.5B under extreme memory constraints (1.0 KB cache, an $86.0\times$ byte compression ratio), Windowed Dual-Space maintains an attention cosine similarity of $0.4779$, outperforming H2O ($0.3210$) by \textbf{+15.69\%}, while ensuring 100\% factual retention in Needle-In-A-Haystack benchmarks where eviction methods completely discard the target information.
\end{abstract}

\section{Phase Dispersion Theorem}
\begin{theorem}[Magnitude Collapse under RoPE Averaging]
Let $u \in \mathbb{R}^{d_k}$ be a canonical key vector. Suppose $N$ tokens sharing identical semantic content $u$ appear at distinct positions $\{p_1, \dots, p_N\}$, generating physical keys $k_n = R(p_n) u$. Let $\bar{k} = \frac{1}{N} \sum_{n=1}^N k_n$.
Then:
\begin{equation}
\mathbb{E}[\|\bar{k}\|^2] = \frac{\|u\|^2}{N} \xrightarrow{N \to \infty} 0
\end{equation}
\end{theorem}

\begin{proof}
For each 2D sub-vector $u^{(m)} \in \mathbb{R}^2$, the inner product is $\langle k_i^{(m)}, k_j^{(m)} \rangle = \|u^{(m)}\|^2 \cos((p_i - p_j) \theta_m)$.
Integrating over distance $\Delta = p_i - p_j$:
\begin{equation}
\mathbb{E}_\Delta [\cos(\Delta \theta_m)] = \frac{\sin(L \theta_m)}{L \theta_m} \xrightarrow{L \theta_m \gg 1} 0
\end{equation}
Taking the expectation of $\|\bar{k}\|^2$:
\begin{equation}
\mathbb{E}[\|\bar{k}\|^2] = \frac{1}{N^2} \sum_{i=1}^N \|k_i\|^2 + \frac{1}{N^2} \sum_{i \ne j} \mathbb{E}[k_i^\top k_j] = \frac{\|u\|^2}{N} + 0 = \frac{\|u\|^2}{N} \to 0
\end{equation}
As $N$ grows, the centroid norm collapses to zero.
\end{proof}

\section{The Windowed Dual-Space Formulation}
\subsection{Canonical De-RoPE Projection}
\begin{equation}
u_t = R(-t) k_t
\end{equation}
In $U$-space, $\cos(u_i, u_j) \equiv 1.0$ for identical semantic tokens across all positions.

\subsection{Windowed Position-Constrained Clustering}
The context is partitioned into windows of length $W \le 32$. Inside each window, canonical keys $u$ and values $v$ are clustered into centroids. The cluster position $\bar{p}_c = \frac{1}{\pi_c} \sum_{i \in c} p_i$ is physically exact.
The physical centroid is reconstructed as:
\begin{equation}
\bar{k}_c = R(\bar{p}_c) \bar{u}_c
\end{equation}
Because $R(\bar{p}_c)$ is orthogonal, $\|\bar{k}_c\| = \|\bar{u}_c\| \approx \|u\|$, completely preventing norm collapse.

\subsection{Inference Evaluation}
The cluster attention logit is computed as:
\begin{equation}
\tilde{z}_c = \frac{q_t^\top \bar{k}_c}{\sqrt{d_k}} + \ln \pi_c
\end{equation}

\section{Byte-for-Byte Memory Analysis}
A standard token requires $(64 + 64) \times 2 = 256$ bytes in FP16.
A Windowed Dual-Space centroid stores $\{\bar{u}_c, \bar{v}_c, \bar{p}_c, \pi_c\}$, requiring $(128 + 2) \times 2 = 260$ bytes ($1.01\times$ of a vanilla token).
No dense matrices are stored.

\section{Conclusion}
Windowed Dual-Space Centroid KV resolves the RoPE phase dispersion barrier without hidden memory bloat. At 260 bytes per centroid, it provides a principled, non-lossy alternative to token eviction under extreme memory constraints. Across both Qwen and Llama flagship families at $>30\times$ compression, it achieves 100\% factual retrieval fidelity where prior eviction methods completely fail.
