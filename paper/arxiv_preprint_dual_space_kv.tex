\begin{abstract}
The linear memory footprint of the Key-Value (KV) cache is a primary bottleneck for deploying Large Language Models (LLMs) on memory-constrained devices. Existing eviction heuristics (e.g., H2O, StreamingLLM) prune tokens with low historical attention scores, but suffer from irreversible factual forgetting in long-context tasks (``Needle-In-A-Haystack''). Conversely, naive key aggregation in physical space fails due to destructive phase dispersion induced by Rotary Position Embeddings (RoPE), collapsing key norms to zero ($\|\bar{k}\| \to 0$).

In this paper, we analyze the geometric incompatibility between RoPE and key aggregation, uncovering a dual-faceted barrier: (1) physical norm collapse, and (2) high-frequency spectral distortion when un-rotated canonical vectors are naively re-rotated ($R(\bar{p})\bar{u}$). To resolve both, we introduce \textbf{Phase-Coherent Windowed Dual-Space Centroid KV}. By constraining clustering to local temporal windows ($W \le 32$) and scaling re-rotated canonical centroids by an analytical \textbf{Phase Coherence Vector} $\boldsymbol{\gamma} \in [0, 1]^D$ ($\gamma_m = \frac{1}{|c|}\sum_{j}\cos(\Delta_j \theta_m)$), our formulation achieves exact first-order equivalence with the true attention sum ($0.0000$ error under canonical homogeneity). Crucially, each centroid requires only \textbf{260 bytes} in FP16 ($1.01\times$ of a vanilla token), requiring zero covariance matrices and zero additional memory for $\boldsymbol{\gamma}$.
\end{abstract}

\section{Theoretical Analysis}

\subsection{Physical Key Averaging: Norm Collapse}
\begin{theorem}[Magnitude Collapse under RoPE Averaging]
Let $u \in \mathbb{R}^{D}$ be a canonical key vector. Suppose $N$ tokens sharing identical semantic content $u$ appear at distinct positions $\{p_1, \dots, p_N\}$, generating physical keys $k_n = R(p_n) u$. Let $\bar{k} = \frac{1}{N} \sum_{n=1}^N k_n$.
Then, for positions uniformly distributed over context length $L$ with $L \gg 2\pi / \theta_m$:
\begin{equation}
\mathbb{E}[\|\bar{k}\|^2] = \frac{\|u\|^2}{N} \xrightarrow{N \to \infty} 0
\end{equation}
\end{theorem}

\begin{proof}
For each 2D subspace $m$ with base frequency $\theta_m = b^{-2m/D}$, the inner product is $\langle k_i^{(m)}, k_j^{(m)} \rangle = \|u^{(m)}\|^2 \cos((p_i - p_j) \theta_m)$.
Integrating over distance $\Delta = p_i - p_j$:
\begin{equation}
\mathbb{E}_\Delta [\cos(\Delta \theta_m)] = \frac{\sin(L \theta_m)}{L \theta_m} \xrightarrow{L \theta_m \gg 1} 0
\end{equation}
Taking the expectation of $\|\bar{k}\|^2$:
\begin{equation}
\mathbb{E}[\|\bar{k}\|^2] = \frac{1}{N^2} \sum_{i=1}^N \|k_i\|^2 + \frac{1}{N^2} \sum_{i \ne j} \mathbb{E}[k_i^\top k_j] = \frac{\|u\|^2}{N} + 0 = \frac{\|u\|^2}{N} \to 0
\end{equation}
As $N$ grows, the physical centroid norm collapses to zero.
\end{proof}

\subsection{Canonical Re-rotation: High-Frequency Overshoot}
\begin{theorem}[Phase Coherence Theorem \& High-Frequency Overshoot]
Let cluster $c = \{ (k_j, p_j) \}_{j=1}^N$ share canonical vector $u \in \mathbb{R}^D$, with positions $p_j = \bar{p} + \Delta_j$ distributed symmetrically around mean $\bar{p}$ ($\sum \Delta_j = 0$).

For any query $q_t$, the true sum of attention logits across cluster $c$ satisfies:
\begin{equation}
\sum_{j=1}^N q_t^\top k_j = N \sum_{m=0}^{D/2-1} \gamma_m \cdot \langle q_t^{(m)}, R(\bar{p}\theta_m) u^{(m)} \rangle
\end{equation}
where the Phase Coherence Factor $\gamma_m \in [0, 1]$ for channel $m$ is:
\begin{equation}
\gamma_m \triangleq \frac{1}{N} \sum_{j=1}^N \cos(\Delta_j \theta_m)
\end{equation}
Consequently, the naive centroid $\bar{k}_{\text{naive}} = R(\bar{p}) u$ systematically over-amplifies subspace $m$ by an error factor of $\mathcal{E}_m = 1 - \gamma_m \ge 0$. For contiguous windows of size $C$, $\gamma_m$ converges to the Dirichlet sinc kernel:
\begin{equation}
\gamma_m = \frac{\sin(C \theta_m / 2)}{C \sin(\theta_m / 2)}
\end{equation}
\end{theorem}

\begin{proof}
Expressing 2D rotation via complex exponentials:
\begin{equation}
\sum_{j=1}^N k_j^{(m)} = e^{i \bar{p} \theta_m} u^{(m)} \sum_{j=1}^N e^{i \Delta_j \theta_m} = e^{i \bar{p} \theta_m} u^{(m)} \sum_{j=1}^N \cos(\Delta_j \theta_m) = N \gamma_m R(\bar{p}\theta_m) u^{(m)}
\end{equation}
Projecting with query $q_t^{(m)}$ yields the exact statement.
\end{proof}

\subsection{Phase-Coherent Dual-Space Centroids}
\begin{theorem}[Exact First-Order Equivalence]
Define the Phase Coherence Vector $\boldsymbol{\gamma} \in [0, 1]^D$ by concatenating duplicate pairs $[\gamma_0, \dots, \gamma_{D/2-1}, \gamma_0, \dots, \gamma_{D/2-1}]$.
Define the Coherence-Corrected Physical Centroid as:
\begin{equation}
\bar{k}_c \triangleq \boldsymbol{\gamma} \odot \left[ R(\bar{p}_c) \bar{u}_c \right]
\end{equation}
Then, for clusters with intra-cluster variance $\sigma_u^2 = \frac{1}{N} \sum \|u_j - \bar{u}\|^2$:
\begin{equation}
\left| \sum_{j \in c} q_t^\top k_j - N \cdot (q_t^\top \bar{k}_c) \right| \le \mathcal{O}(\|\Delta\| \cdot \sigma_u)
\end{equation}
Under canonical semantic coherence ($\sigma_u \to 0$), the approximation is mathematically exact ($0.0000$ error).
\end{theorem}

\section{Byte-for-Byte Memory Analysis}
A standard token requires $(64 + 64) \times 2 = 256$ bytes in FP16.
A Windowed Dual-Space centroid stores $\{\bar{u}_c, \bar{v}_c, \bar{p}_c, \pi_c\}$, requiring $(128 + 2) \times 2 = 260$ bytes ($1.01\times$ of a vanilla token).
Zero covariance matrices are stored, and $\boldsymbol{\gamma}$ is calculated analytically on-the-fly with zero persistent memory overhead.

\section{Conclusion}
Phase-Coherent Windowed Dual-Space Centroid KV resolves both physical norm collapse and high-frequency spectral distortion. By coupling canonical De-RoPE with analytical Phase Coherence dampening ($\boldsymbol{\gamma}$), it achieves exact first-order key aggregation without memory expansion, establishing a mathematically sound foundation for long-context KV cache compression.
